Create API endpoint: GET /api/deals
  Query: SELECT l.*, dm.avgPricePsf as districtPsf
         FROM Listing l
         JOIN DistrictMetrics dm ON l.district = dm.district
         WHERE l.pricePsf < (dm.avgPricePsf * 0.88)
           AND dm.period = '30d' AND l.isActive = true
         ORDER BY ((dm.avgPricePsf - l.pricePsf) / dm.avgPricePsf) DESC
         LIMIT 10
